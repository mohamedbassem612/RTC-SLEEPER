cmake_minimum_required(VERSION 3.22)
project(RTCSleeper VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test)
add_library(sleeper-core src/engine.cpp src/client.cpp)
target_include_directories(sleeper-core PUBLIC src)
target_link_libraries(sleeper-core PUBLIC Qt6::Core)
target_compile_options(sleeper-core PRIVATE -Wall -Wextra -Wpedantic)
add_library(sleeper-ui src/window.cpp)
target_link_libraries(sleeper-ui PUBLIC sleeper-core Qt6::Widgets)
add_executable(sleep-scheduler-engine src/main.cpp)
target_compile_definitions(sleep-scheduler-engine PRIVATE RTC_HEADLESS)
target_link_libraries(sleep-scheduler-engine PRIVATE sleeper-core)
add_executable(sleep-scheduler src/main.cpp)
target_link_libraries(sleep-scheduler PRIVATE sleeper-ui)
enable_testing()
add_executable(test-engine tests/test_engine.cpp)
target_link_libraries(test-engine PRIVATE sleeper-core Qt6::Test)
add_test(NAME engine COMMAND test-engine)
add_executable(test-interface tests/test_interface.cpp)
target_link_libraries(test-interface PRIVATE sleeper-ui Qt6::Test)
add_test(NAME interface COMMAND test-interface)
set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
install(TARGETS sleep-scheduler sleep-scheduler-engine RUNTIME DESTINATION bin)
